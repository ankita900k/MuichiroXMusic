async def store_command(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):

    if not update.message:
        return

    if getattr(
        update.message,
        "sender_chat",
        None
    ):

        await reject_anonymous_user(
            update
        )

        return

    user = update.effective_user

    if not user:
        return

    if user.is_bot:
        return

    await ensure_player(user)

    try:

        async with db_lock:

            cursor.execute("""
                SELECT points
                FROM slayer_points
                WHERE user_id = %s
                AND is_bot = FALSE
            """, (
                user.id,
            ))

            row = cursor.fetchone()

        coins = (
            row[0]
            if row
            else 0
        )

    except Exception as e:

        print(
            f"❌ Store balance error: {e}"
        )

        return


    lines = []

    for item_id, item in STORE_ITEMS.items():

        lines.append(
            f"<b>{item_id}.</b> "
            f"{item['emoji']} <b>{html.escape(item['name'])}</b>\n"
            f"   💎 <b>{item['price']:,}</b> coins\n"
            f"   <i>{html.escape(item['description'])}</i>"
        )


    item_catalog = hunt_quote(
        "\n\n".join(lines)
    )

    text = (
        "╭━━━〔 🏪 <b>𝗦𝗟𝗔𝗬𝗘𝗥 𝗦𝗧𝗢𝗥𝗘</b> 〕━━━╮\n\n"
        f"💎 <b>Your Coins:</b> {coins:,}\n\n"
        + item_catalog
        + "\n\n"
        "━━━━━━━━━━━━━━━━━━━━\n"
        "🛒 <b>BUY:</b> <code>/buy &lt;item id&gt;</code>\n"
        "🎒 <b>INVENTORY:</b> <code>/inventory</code>\n"
        "🎖️ <b>EQUIP:</b> <code>/equipbadge &lt;item id&gt;</code>\n"
        "━━━━━━━━━━━━━━━━━━━━"
    )

    await update.message.reply_text(
        text,
        parse_mode=ParseMode.HTML
    )
